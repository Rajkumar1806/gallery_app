    _request();
  }

    final state = await PhotoManager.requestPermissionExtend(
      requestOption: const PermissionRequestOption(
        androidPermission: AndroidPermission(
          type: RequestType.common,
          mediaLocation: true,
        ),
      ),
    );
    if (!mounted) return;
    setState(() => _state = state);
    if (state.hasAccess) {
      GalleryScope.of(context, listen: false).load();
    }

  Widget build(BuildContext context) {
    final state = _state;
    if (state != null && state.hasAccess) return const HomeShell();

final _dayFmt = DateFormat('MMM d, yyyy');
final _dayFmtThisYear = DateFormat('EEE, MMM d');
final _monthFmt = DateFormat('MMMM yyyy');

String dayLabel(DateTime d) {
  final now = DateTime.now();
  final today = DateTime(now.year, now.month, now.day);
  final day = DateTime(d.year, d.month, d.day);
  final diff = today.difference(day).inDays;
  if (diff == 0) return 'Today';
  if (diff == 1) return 'Yesterday';
  return _dayFmt.format(d);
}

String dayLabelLong(DateTime d) {
  final now = DateTime.now();
  final s = dayLabel(d);
  if (s == 'Today' || s == 'Yesterday') return s;
  return d.year == now.year ? _dayFmtThisYear.format(d) : _dayFmt.format(d);
}

/// Assumes [assets] is sorted newest first.
List<AssetGroup> groupByDay(List<AssetEntity> assets, {bool long = false}) {
  final out = <AssetGroup>[];
  DateTime? current;
  List<AssetEntity>? bucket;
  for (final a in assets) {
    final d = a.createDateTime;
    final key = DateTime(d.year, d.month, d.day);
    if (key != current) {
      current = key;
      bucket = [];
      out.add(AssetGroup(long ? dayLabelLong(d) : dayLabel(d), bucket));
    }
    bucket!.add(a);
  }
  return out;
}

List<AssetGroup> groupByMonth(List<AssetEntity> assets) {
  final out = <AssetGroup>[];
  DateTime? current;
  List<AssetEntity>? bucket;
  for (final a in assets) {
    final d = a.createDateTime;
    final key = DateTime(d.year, d.month);
    if (key != current) {
      current = key;
      bucket = [];
      out.add(AssetGroup(_monthFmt.format(d), bucket));
    }
    bucket!.add(a);
  }
  return [
    for (final g in out) AssetGroup(g.title, g.assets, trailing: '${g.assets.length} items'),
  ];
}

Map<int, List<AssetEntity>> groupByYear(List<AssetEntity> assets) {
  final out = <int, List<AssetEntity>>{};
  for (final a in assets) {
    (out[a.createDateTime.year] ??= []).add(a);
  }
  return out;
}

String countLabel(List<AssetEntity> assets) {
  final videos = assets.where((a) => a.type == AssetType.video).length;
  final photos = assets.length - videos;
  final parts = <String>[
    if (photos > 0) '$photos photo${photos == 1 ? '' : 's'}',
    if (videos > 0) '$videos video${videos == 1 ? '' : 's'}',
  ];
  return parts.isEmpty ? 'Empty' : parts.join(', ');

/// Single source of truth for everything the UI shows.
class MediaRepository extends ChangeNotifier {
  MediaRepository() {
    scanner = SmartScanner(this);
  }

  late final SmartScanner scanner;

  bool loading = false;
  bool loaded = false;
  List<AssetEntity> all = const [];
  List<AlbumInfo> albums = const [];

  /// asset id -> album name (used by search).
  final Map<String, String> _albumOf = {};
  Set<String> _favorites = {};
  List<Memory>? _memories;
  Timer? _reloadDebounce;
  bool _listening = false;

  // ---------------------------------------------------------------- loading

  Future<void> load() async {
    if (loading) return;
    loading = true;
    notifyListeners();

    final prefs = await SharedPreferences.getInstance();
    _favorites = (prefs.getStringList('favorites') ?? const []).toSet();

    final filter = FilterOptionGroup(
      orders: [const OrderOption(type: OrderOptionType.createDate, asc: false)],
    );
    final paths = await PhotoManager.getAssetPathList(
      type: RequestType.common,
      hasAll: true,
      filterOption: filter,
    );

    final allAssets = <AssetEntity>[];
    final albumList = <AlbumInfo>[];
    _albumOf.clear();

    for (final path in paths) {
      final assets = await _readAll(path);
      if (path.isAll) {
        allAssets.addAll(assets);
      } else if (assets.isNotEmpty) {
        albumList.add(AlbumInfo(id: path.id, name: path.name, assets: assets));
        for (final a in assets) {
          _albumOf[a.id] = path.name;
        }
      }
    }

    // Biggest albums first, but keep the camera roll on top like ColorOS.
    albumList.sort((a, b) {
      final ac = _isCamera(a.name), bc = _isCamera(b.name);
      if (ac != bc) return ac ? -1 : 1;
      return b.count.compareTo(a.count);
    });

    all = allAssets;
    albums = albumList;
    _memories = null;
    loading = false;
    loaded = true;
    notifyListeners();

    _listenForChanges();
    unawaited(scanner.start());
  }

  Future<List<AssetEntity>> _readAll(AssetPathEntity path) async {
    final total = await path.assetCountAsync;
    const page = 1000;
    final out = <AssetEntity>[];
    for (var start = 0; start < total; start += page) {
      final end = (start + page).clamp(0, total);
      out.addAll(await path.getAssetListRange(start: start, end: end));
    }
    return out;
  }

  void _listenForChanges() {
    if (_listening) return;
    _listening = true;
    PhotoManager.addChangeCallback(_onMediaChanged);
    PhotoManager.startChangeNotify();
  }

  void _onMediaChanged(MethodCall _) {
    _reloadDebounce?.cancel();
    _reloadDebounce = Timer(const Duration(seconds: 1), () {
      loading = false;
      load();
    });
  }

  static bool _isCamera(String name) {
    final n = name.toLowerCase();
    return n == 'camera' || n == 'dcim';
  }

  // ------------------------------------------------------------- favorites

  bool isFavorite(AssetEntity a) => _favorites.contains(a.id);

  Future<void> toggleFavorite(AssetEntity a) async {
    if (!_favorites.remove(a.id)) _favorites.add(a.id);
    notifyListeners();
    final prefs = await SharedPreferences.getInstance();
    await prefs.setStringList('favorites', _favorites.toList());
  }

  // ---------------------------------------------------------------- delete

  /// Deletes from the device. Android 11+ shows the system confirmation.
  Future<bool> delete(List<AssetEntity> items) async {
    final ids = items.map((e) => e.id).toList();
    final deleted = (await PhotoManager.editor.deleteWithIds(ids)).toSet();
    if (deleted.isEmpty) return false;
    all = all.where((a) => !deleted.contains(a.id)).toList();
    albums = [
      for (final al in albums)
        AlbumInfo(
          id: al.id,
          name: al.name,
          assets: al.assets.where((a) => !deleted.contains(a.id)).toList(),
        ),
    ].where((al) => al.count > 0).toList();
    _favorites.removeAll(deleted);
    _memories = null;
    notifyListeners();
    return true;
  }

  // ------------------------------------------------------------ media types

  String? albumNameOf(AssetEntity a) => _albumOf[a.id];

  static bool isGif(AssetEntity a) =>
      (a.mimeType?.toLowerCase() == 'image/gif') ||
      (a.title?.toLowerCase().endsWith('.gif') ?? false);

  bool _isScreenshot(AssetEntity a) {
    final album = _albumOf[a.id]?.toLowerCase() ?? '';
    final title = a.title?.toLowerCase() ?? '';
    return album.contains('screenshot') || title.startsWith('screenshot');
  }

  static bool _isPanorama(AssetEntity a) {
    if (a.type != AssetType.image || a.width == 0 || a.height == 0) return false;
    final r = a.width / a.height;
    return r >= 2.2 || (r <= 1 / 2.2 && a.height > 4000);
  }

  List<AssetEntity> mediaType(MediaType t) => switch (t) {
        MediaType.videos => all.where((a) => a.type == AssetType.video).toList(),
        MediaType.gifs => all.where(isGif).toList(),
        MediaType.screenshots => all.where(_isScreenshot).toList(),
        MediaType.favorites => all.where(isFavorite).toList(),
        MediaType.panoramas => all.where(_isPanorama).toList(),
        MediaType.recent => all
            .where((a) => DateTime.now().difference(a.createDateTime).inDays <= 30)
            .toList(),
      };

  // -------------------------------------------------------------- memories

  List<Memory> get memories =>
      _memories ??= buildMemories(all, faceScore: scanner.faceCount);

  void invalidateMemories() {
    _memories = null;
    notifyListeners();
  }

  // ---------------------------------------------------------------- search

  static final _monthFmt = DateFormat('MMMM yyyy');
  static final _dayFmt = DateFormat('d MMMM yyyy');
  static final _shortMonth = DateFormat('MMM');
  static final _weekday = DateFormat('EEEE');

  /// Matches file names, album names, dates ("sep 2024", "2025", "monday"),
  /// place names and smart tags.
  List<AssetEntity> search(String raw) {
    final terms = raw.toLowerCase().trim().split(RegExp(r'\s+'))
      ..removeWhere((t) => t.isEmpty);
    if (terms.isEmpty) return const [];

    return all.where((a) {
      final d = a.createDateTime;
      final haystack = StringBuffer()
        ..write(a.title ?? '')
        ..write(' ')
        ..write(_albumOf[a.id] ?? '')
        ..write(' ')
        ..write(_monthFmt.format(d))
        ..write(' ')
        ..write(_dayFmt.format(d))
        ..write(' ')
        ..write(_shortMonth.format(d))
        ..write(' ')
        ..write(_weekday.format(d))
        ..write(' ')
        ..write(a.type == AssetType.video ? 'video videos' : 'photo photos image')
        ..write(' ')
        ..write(scanner.placeNameOf(a.id) ?? '')
        ..write(' ')
        ..write(scanner.tagsOf(a.id).join(' '))
        ..write(isFavorite(a) ? ' favorite favourites' : '');
      final h = haystack.toString().toLowerCase();
      return terms.every(h.contains);
    }).toList();
  }

  @override
  void dispose() {
    _reloadDebounce?.cancel();
    if (_listening) {
      PhotoManager.removeChangeCallback(_onMediaChanged);
      PhotoManager.stopChangeNotify();
    }
    scanner.dispose();
    super.dispose();
  }
}

/// An auto-generated story such as "Two weeks in review – September 2026".
class Memory {
  Memory({
    required this.id,
    required this.title,
    required this.subtitle,
    required this.assets,
    required this.cover,
    required this.start,
  });

  final String id;
  final String title;
  final String subtitle;
  final List<AssetEntity> assets;
  final AssetEntity cover;
  final DateTime start;
}

/// Groups the library into half-month windows ("Two weeks in review") and
/// adds an "On this day" memory when older photos share today's date.
List<Memory> buildMemories(
  List<AssetEntity> all, {
  required int Function(String id) faceScore,
  int minPhotos = 6,
}) {
  final monthFmt = DateFormat('MMMM yyyy');
  final windows = <DateTime, List<AssetEntity>>{};
  for (final a in all) {
    final d = a.createDateTime;
    final start = DateTime(d.year, d.month, d.day <= 14 ? 1 : 15);
    (windows[start] ??= []).add(a);
  }

  final memories = <Memory>[];

  // On this day
  final now = DateTime.now();
  final onThisDay = all
      .where((a) =>
          a.createDateTime.month == now.month &&
          a.createDateTime.day == now.day &&
          a.createDateTime.year < now.year)
      .toList();
  if (onThisDay.length >= 3) {
    final years = now.year - onThisDay.first.createDateTime.year;
    memories.add(Memory(
      id: 'otd-${now.month}-${now.day}',
      title: 'On this day',
      subtitle: years == 1 ? '1 year ago' : '$years years ago',
      assets: onThisDay,
      cover: _pickCover(onThisDay, faceScore),
      start: onThisDay.first.createDateTime,
    ));
  }

  final keys = windows.keys.toList()..sort((a, b) => b.compareTo(a));
  for (final k in keys) {
    final list = windows[k]!;
    if (list.length < minPhotos) continue;
    memories.add(Memory(
      id: 'tw-${k.toIso8601String()}',
      title: 'Two weeks in review',
      subtitle: monthFmt.format(k),
      assets: list,
      cover: _pickCover(list, faceScore),
      start: k,
    ));
  }
  return memories;
}

/// Prefers a photo (not video) with people in it, otherwise the middle one.
AssetEntity _pickCover(List<AssetEntity> list, int Function(String) faceScore) {
  final photos = list.where((a) => a.type == AssetType.image).toList();
  if (photos.isEmpty) return list.first;
  AssetEntity? best;
  var bestScore = 0;
  for (final p in photos) {
    final s = faceScore(p.id);
    if (s > bestScore && s <= 4) {
      best = p;
      bestScore = s;
    }
  }
  return best ?? photos[photos.length ~/ 2];
}

/// Runs ML Kit image labelling + face detection on downscaled thumbnails,
/// entirely on-device, and caches the results.
class SmartScanner extends ChangeNotifier {
  SmartScanner(this.repo);

  final MediaRepository repo;

  /// Newest N photos scanned. Raise it if you want a deeper index.
  static const maxScan = 3000;
  static const _prefsKey = 'smart_index_v1';
  static const _placesKey = 'place_names_v1';

  final Map<String, SmartInfo> _index = {};
  final Map<String, String> _placeNames = {};
  List<PlaceInfo> places = const [];
  Map<String, List<AssetEntity>> tagged = const {};
  List<AssetEntity> withFaces = const [];

  bool running = false;
  int done = 0;
  int total = 0;
  bool _cancel = false;
  bool _restored = false;

  double get progress => total == 0 ? 1 : done / total;

  int faceCount(String id) => _index[id]?.faces ?? 0;
  SmartInfo? infoOf(String id) => _index[id];
  List<String> tagsOf(String id) => _index[id]?.tags ?? const [];

  String? placeNameOf(String id) {
    final info = _index[id];
    if (info?.lat == null) return null;
    return _placeNames[_placeKey(info!.lat!, info.lng!)];
  }

  Future<void> start() async {
    if (running) return;
    running = true;
    _cancel = false;

    final prefs = await SharedPreferences.getInstance();
    if (!_restored) {
      _restored = true;
      final raw = prefs.getString(_prefsKey);
      if (raw != null) {
        final m = jsonDecode(raw) as Map<String, dynamic>;
        m.forEach((k, v) => _index[k] = SmartInfo.fromJson(v as Map<String, dynamic>));
      }
      final rawPlaces = prefs.getString(_placesKey);
      if (rawPlaces != null) {
        _placeNames.addAll((jsonDecode(rawPlaces) as Map).cast<String, String>());
      }
    }
    _rebuild();

    final todo = repo.all
        .where((a) => a.type == AssetType.image)
        .take(maxScan)
        .where((a) => !_index.containsKey(a.id))
        .toList();
    total = todo.length;
    done = 0;
    notifyListeners();

    if (todo.isNotEmpty) {
      final labeler =
          ImageLabeler(options: ImageLabelerOptions(confidenceThreshold: 0.6));
      final faceDetector = FaceDetector(
        options: FaceDetectorOptions(performanceMode: FaceDetectorMode.fast),
      );
      final tmpDir = await getTemporaryDirectory();
      final tmp = File('${tmpDir.path}/scan_thumb.jpg');

      for (final asset in todo) {
        if (_cancel) break;
        try {
          _index[asset.id] = await _scanOne(asset, labeler, faceDetector, tmp);
        } catch (e) {
          debugPrint('scan failed for ${asset.id}: $e');
          _index[asset.id] = SmartInfo(tags: const [], faces: 0);
        }
        done++;
        if (done % 40 == 0 || done == total) {
          _rebuild();
          await prefs.setString(_prefsKey, jsonEncode(_index));
          notifyListeners();
        }
      }
      await labeler.close();
      await faceDetector.close();
      await prefs.setString(_prefsKey, jsonEncode(_index));
    }

    await _resolvePlaceNames(prefs);
    _rebuild();
    running = false;
    repo.invalidateMemories();
    notifyListeners();
  }

  Future<SmartInfo> _scanOne(
    AssetEntity asset,
    ImageLabeler labeler,
    FaceDetector faceDetector,
    File tmp,
  ) async {
    final bytes =
        await asset.thumbnailDataWithSize(ThumbnailSize(640, 640), quality: 85);
    final tags = <String>{};
    var faces = 0;
    double fx = 0.5, fy = 0.4;

    if (bytes != null) {
      await tmp.writeAsBytes(bytes, flush: true);
      final input = InputImage.fromFilePath(tmp.path);

      final labels = await labeler.processImage(input);
      for (final l in labels) {
        final name = l.label.toLowerCase();
        final words = name.split(RegExp(r'[\s-]+'));
        kTagRules.forEach((tag, rules) {
          if (rules.any((w) => name == w || words.contains(w))) tags.add(tag);
        });
      }

      final found = await faceDetector.processImage(input);
      faces = found.length;
      if (found.isNotEmpty) {
        final biggest = found.reduce((a, b) =>
            a.boundingBox.width * a.boundingBox.height >=
                    b.boundingBox.width * b.boundingBox.height
                ? a
                : b);
        final size = await _imageSize(bytes);
        if (size != null) {
          fx = (biggest.boundingBox.center.dx / size.width).clamp(0.0, 1.0);
          fy = (biggest.boundingBox.center.dy / size.height).clamp(0.0, 1.0);
        }
        // A single big face close to the camera is very likely a selfie.
        final rel = biggest.boundingBox.width / (size?.width ?? 640);
        if (faces == 1 && rel > 0.35) tags.add('Selfies');
      }
    }

    double? lat, lng;
    try {
      final ll = await asset.latlngAsync();
      final la = ll?.latitude, lo = ll?.longitude;
      if (la != null && lo != null && !(la == 0 && lo == 0)) {
        lat = la;
        lng = lo;
      }
    } catch (_) {}

    return SmartInfo(
        tags: tags.toList(), faces: faces, fx: fx, fy: fy, lat: lat, lng: lng);
  }

  static Future<ui.Size?> _imageSize(Uint8List bytes) async {
    try {
      final buffer = await ui.ImmutableBuffer.fromUint8List(bytes);
      final desc = await ui.ImageDescriptor.encoded(buffer);
      final s = ui.Size(desc.width.toDouble(), desc.height.toDouble());
      desc.dispose();
      buffer.dispose();
      return s;
    } catch (_) {
      return null;
    }
  }

  static String _placeKey(double lat, double lng) =>
      '${(lat * 10).round()}:${(lng * 10).round()}';

  Future<void> _resolvePlaceNames(SharedPreferences prefs) async {
    final keys = <String, SmartInfo>{};
    for (final info in _index.values) {
      if (info.lat == null) continue;
      keys.putIfAbsent(_placeKey(info.lat!, info.lng!), () => info);
    }
    var changed = false;
    for (final e in keys.entries) {
      if (_placeNames.containsKey(e.key) || _cancel) continue;
      try {
        final marks = await placemarkFromCoordinates(e.value.lat!, e.value.lng!);
        if (marks.isNotEmpty) {
          final p = marks.first;
          final name = [p.subLocality, p.locality, p.subAdministrativeArea, p.administrativeArea]
              .firstWhere((s) => s != null && s.trim().isNotEmpty, orElse: () => null);
          _placeNames[e.key] = name ?? 'Unknown place';
          changed = true;
        }
      } catch (_) {
        // Offline or geocoder unavailable – try again next launch.
      }
    }
    if (changed) await prefs.setString(_placesKey, jsonEncode(_placeNames));
  }

  void _rebuild() {
    final tagMap = <String, List<AssetEntity>>{};
    final faces = <AssetEntity>[];
    final placeMap = <String, List<AssetEntity>>{};

    for (final a in repo.all) {
      final info = _index[a.id];
      if (info == null) continue;
      for (final t in info.tags) {
        (tagMap[t] ??= []).add(a);
      }
      if (info.faces > 0) faces.add(a);
      if (info.lat != null) {
        (placeMap[_placeKey(info.lat!, info.lng!)] ??= []).add(a);
      }
    }

    // Merge clusters that resolved to the same place name.
    final named = <String, List<AssetEntity>>{};
    placeMap.forEach((key, list) {
      final name = _placeNames[key] ?? 'Locating…';
      (named[name] ??= []).addAll(list);
    });
    places = named.entries
        .map((e) => PlaceInfo(key: e.key, name: e.key, assets: e.value
          ..sort((a, b) => b.createDateTime.compareTo(a.createDateTime))))
        .toList()
      ..sort((a, b) => b.assets.length.compareTo(a.assets.length));

    tagged = Map.fromEntries(tagMap.entries.toList()
      ..sort((a, b) => b.value.length.compareTo(a.value.length)));
    withFaces = faces;
  }

  @override
  void dispose() {
    _cancel = true;
    super.dispose();
  }
}

/// A thumbnail for one photo/video with video-duration and favourite badges.
class AssetThumb extends StatelessWidget {
  const AssetThumb(
    this.asset, {
    super.key,
    this.radius = 0,
    this.size = 300,
    this.alignment = Alignment.center,
    this.showBadges = true,
  });

  final AssetEntity asset;
  final double radius;
  final int size;
  final Alignment alignment;
  final bool showBadges;

  @override
  Widget build(BuildContext context) {
    final image = Image(
      image: AssetEntityImageProvider(
        asset,
        isOriginal: false,
        thumbnailSize: ThumbnailSize.square(size),
      ),
      fit: BoxFit.cover,
      alignment: alignment,
      gaplessPlayback: true,
      filterQuality: FilterQuality.medium,
      errorBuilder: (_, __, ___) => Container(
        color: Colors.black12,
        child: const Icon(Icons.broken_image_outlined, color: Colors.black26),
      ),
      frameBuilder: (context, child, frame, sync) => sync
          ? child
          : AnimatedOpacity(
              opacity: frame == null ? 0 : 1,
              duration: const Duration(milliseconds: 180),
              child: child,
            ),
    );

    Widget content = Stack(
      fit: StackFit.expand,
      children: [
        ColoredBox(color: Colors.black.withOpacity(0.05)),
        image,
        if (showBadges && asset.type == AssetType.video)
          Positioned(
            right: 5,
            bottom: 4,
            child: Row(children: [
              const Icon(Icons.play_arrow_rounded, color: Colors.white, size: 15,
                  shadows: [Shadow(blurRadius: 4)]),
              Text(formatDuration(asset.videoDuration),
                  style: const TextStyle(
                      color: Colors.white,
                      fontSize: 11,
                      fontWeight: FontWeight.w600,
                      shadows: [Shadow(blurRadius: 4)])),
            ]),
          ),
        if (showBadges && GalleryScope.of(context).isFavorite(asset))
          const Positioned(
            left: 5,
            bottom: 4,
            child: Icon(Icons.favorite, color: Colors.white, size: 14,
                shadows: [Shadow(blurRadius: 4)]),
          ),
      ],
    );
    if (radius > 0) {
      content = ClipRRect(borderRadius: BorderRadius.circular(radius), child: content);
    }
    return content;
  }
}

String formatDuration(Duration d) {
  final h = d.inHours;
  final m = d.inMinutes.remainder(60).toString().padLeft(h > 0 ? 2 : 1, '0');
  final s = d.inSeconds.remainder(60).toString().padLeft(2, '0');
  return h > 0 ? '$h:$m:$s' : '$m:$s';
}

/// Builds slivers for date-grouped grids. Tapping opens the viewer over the
/// flattened list so you can swipe across groups.
List<Widget> groupedGridSlivers(
  BuildContext context, {
  required List<AssetGroup> groups,
  required int columns,
  double spacing = 2,
  double headerHeight = 50,
  TextStyle? headerStyle,
  int thumbSize = 300,
}) {
  final flat = <AssetEntity>[for (final g in groups) ...g.assets];
  final style = headerStyle ??
      Theme.of(context).textTheme.titleSmall?.copyWith(fontWeight: FontWeight.w600);
  final slivers = <Widget>[];
  var offset = 0;
  for (final g in groups) {
    final base = offset;
    slivers.add(SliverToBoxAdapter(
      child: Container(
        height: headerHeight,
        alignment: Alignment.bottomLeft,
        padding: const EdgeInsets.fromLTRB(16, 0, 16, 10),
        child: Row(children: [
          Expanded(child: Text(g.title, style: style)),
          if (g.trailing != null)
            Text(g.trailing!,
                style: TextStyle(
                    fontSize: 12,
                    color: GalleryColors.subtle(Theme.of(context).brightness))),
        ]),
      ),
    ));
    slivers.add(SliverGrid(
      gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
        crossAxisCount: columns,
        mainAxisSpacing: spacing,
        crossAxisSpacing: spacing,
      ),
      delegate: SliverChildBuilderDelegate(
        (context, i) => GestureDetector(
          onTap: () => openViewer(context, flat, base + i),
          child: AssetThumb(g.assets[i], size: thumbSize),
        ),
        childCount: g.assets.length,
      ),
    ));
    offset += g.assets.length;
  }
  return slivers;
}

/// Alignment that centres a thumbnail on the detected face.
Alignment faceAlignment(SmartScanner s, AssetEntity a) {
  final info = s.infoOf(a.id);
  if (info == null || info.faces == 0) return Alignment.center;
  return Alignment(info.fx * 2 - 1, info.fy * 2 - 1);
}

class _HomeShellState extends State<HomeShell> {
  final MediaRepository _repository = MediaRepository();
  List<AssetEntity> _images = [];
  bool _isLoading = true;

  @override
  void initState() {
    super.initState();
    _loadImages();
  }

  Future<void> _loadImages() async {
    final images = await _repository.getRecentImages();
    if (mounted) {
      setState(() {
        _images = images;
        _isLoading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_isLoading) {
      return const Scaffold(body: Center(child: CircularProgressIndicator()));
    }

    return Scaffold(
      appBar: AppBar(title: const Text('Gallery')),
      body: _images.isEmpty 
        ? const Center(child: Text('No images found or permission denied.'))
        : GridView.builder(
            padding: const EdgeInsets.all(2),
            gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
              crossAxisCount: 3,
              crossAxisSpacing: 2,
              mainAxisSpacing: 2,
            ),
            itemCount: _images.length,
            itemBuilder: (context, index) {
              return FutureBuilder<Uint8List?>(
                future: _images[index].thumbnailDataWithSize(const ThumbnailSize(200, 200)),
                builder: (context, snapshot) {
                  if (snapshot.hasData && snapshot.data != null) {
                    return Image.memory(snapshot.data!, fit: BoxFit.cover);
                  }
                  return const Center(child: CircularProgressIndicator());
                },
              );
            },
          ),
    );
  }

class SlideshowScreen extends StatefulWidget {

  final List<AssetEntity> assets;

  @override
  State<SlideshowScreen> createState() => _SlideshowScreenState();

class _SlideshowScreenState extends State<SlideshowScreen> {
  late final List<AssetEntity> _photos =
      widget.assets.where((a) => a.type == AssetType.image).toList().reversed.toList();
  int _index = 0;
  Timer? _timer;
  bool _paused = false;

  @override
  void initState() {
    super.initState();
    _timer = Timer.periodic(const Duration(seconds: 4), (_) {
      if (_paused || _photos.isEmpty) return;
      setState(() => _index = (_index + 1) % _photos.length);
    });
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    if (_photos.isEmpty) {
      return const Scaffold(body: Center(child: Text('No photos to play')));
    }
    final asset = _photos[_index];
    return Scaffold(
      backgroundColor: Colors.black,
      body: GestureDetector(
        onTap: () => setState(() => _paused = !_paused),
        child: Stack(fit: StackFit.expand, children: [
          AnimatedSwitcher(
            duration: const Duration(milliseconds: 900),
            child: TweenAnimationBuilder<double>(
              key: ValueKey(asset.id),
              tween: Tween(begin: 1.0, end: 1.12),
              duration: const Duration(seconds: 5),
              builder: (_, scale, child) => Transform.scale(scale: scale, child: child),
              child: Image(
                image: AssetEntityImageProvider(asset,
                    isOriginal: false, thumbnailSize: ThumbnailSize.square(1600)),
                fit: BoxFit.contain,
              ),
            ),
          ),
          SafeArea(
            child: Align(
              alignment: Alignment.topLeft,
              child: IconButton(
                color: Colors.white,
                icon: const Icon(Icons.close),
                onPressed: () => Navigator.of(context).pop(),
              ),
            ),
          ),
          if (_paused)
            const Center(
                child: Icon(Icons.pause_circle_filled, color: Colors.white70, size: 72)),
          Positioned(
            left: 16,
            right: 16,
            bottom: MediaQuery.paddingOf(context).bottom + 16,
            child: LinearProgressIndicator(
              value: (_index + 1) / _photos.length,
              backgroundColor: Colors.white24,
              color: Colors.white,
              minHeight: 2,
            ),
          ),
        ]),
      ),
    );
  }
}

/// Zoom levels, from densest to largest.
enum Zoom { years, months, all, days }

extension on Zoom {
  String get label => switch (this) {
        Zoom.years => 'Years',
        Zoom.months => 'Months',
        Zoom.days => 'Days',
        Zoom.all => 'All',
      };
  int get columns => switch (this) {
        Zoom.years => 1,
        Zoom.months => 6,
        Zoom.all => 4,
        Zoom.days => 3,
      };
  double get spacing => switch (this) {
        Zoom.months => 1.5,
        Zoom.days => 3,
        _ => 2,
      };
}

class PhotosScreen extends StatefulWidget {
  const PhotosScreen({super.key});

  @override
  State<PhotosScreen> createState() => _PhotosScreenState();
}

class _PhotosScreenState extends State<PhotosScreen> {
  static const _headerHeight = 50.0;
  Zoom _zoom = Zoom.all;
  final _scroll = ScrollController();

  // Pinch tracking (Listener doesn't fight the scroll gesture).
  final Map<int, Offset> _pointers = {};
  double? _startSpan;
  bool _pinchHandled = false;

  @override
  void dispose() {
    _scroll.dispose();
    super.dispose();
  }

  void _setZoom(Zoom z, {int? jumpToGroup}) {
    if (z == _zoom) return;
    final ratio = _scroll.hasClients && _scroll.position.maxScrollExtent > 0
        ? _scroll.offset / _scroll.position.maxScrollExtent
        : 0.0;
    setState(() => _zoom = z);
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!_scroll.hasClients) return;
      final max = _scroll.position.maxScrollExtent;
      final target = jumpToGroup != null ? _offsetOfGroup(jumpToGroup) : ratio * max;
      _scroll.jumpTo(target.clamp(0.0, max));
    });
  }

  /// Offset of group [index] in the months grid (fixed header heights make
  /// this exact without measuring).
  double _offsetOfGroup(int index) {
    final repo = GalleryScope.of(context, listen: false);
    final groups = groupByMonth(repo.all);
    final width = MediaQuery.sizeOf(context).width;
    const z = Zoom.months;
    final cell = (width - z.spacing * (z.columns - 1)) / z.columns;
    var y = 0.0;
    for (var i = 0; i < index && i < groups.length; i++) {
      final rows = (groups[i].assets.length / z.columns).ceil();
      y += _headerHeight + rows * cell + (rows - 1) * z.spacing;
    }
    // Spacer above the first header is (top - 40); keep the header visible
    // just below the floating search bar.
    return math.max(0, y - 40);
  }

  void _onPointerDown(PointerDownEvent e) {
    _pointers[e.pointer] = e.position;
    if (_pointers.length == 2) {
      _startSpan = _span();
      _pinchHandled = false;
    }
  }

  void _onPointerMove(PointerMoveEvent e) {
    if (!_pointers.containsKey(e.pointer)) return;
    _pointers[e.pointer] = e.position;
    if (_pointers.length != 2 || _startSpan == null || _pinchHandled) return;
    final scale = _span() / _startSpan!;
    final idx = _zoom.index;
    if (scale > 1.35 && idx < Zoom.values.length - 1) {
      _pinchHandled = true;
      _setZoom(Zoom.values[idx + 1]);
    } else if (scale < 0.72 && idx > 0) {
      _pinchHandled = true;
      _setZoom(Zoom.values[idx - 1]);
    }
  }

  void _onPointerUp(PointerEvent e) {
    _pointers.remove(e.pointer);
    if (_pointers.length < 2) _startSpan = null;
  }

  double _span() {
    final p = _pointers.values.toList();
    return math.max(1, (p[0] - p[1]).distance);
  }

  @override
  Widget build(BuildContext context) {
    final repo = GalleryScope.of(context);
    final top = ShellInsets.top(context);
    final bottom = ShellInsets.bottom(context);

    if (repo.all.isEmpty) {
      return const Center(child: Text('No photos or videos yet'));
    }

    final slivers = switch (_zoom) {
      Zoom.years => [_yearsSliver(repo.all)],
      Zoom.months => groupedGridSlivers(context,
          groups: groupByMonth(repo.all),
          columns: _zoom.columns,
          spacing: _zoom.spacing,
          headerHeight: _headerHeight,
          thumbSize: 160),
      Zoom.all => groupedGridSlivers(context,
          groups: groupByDay(repo.all),
          columns: _zoom.columns,
          spacing: _zoom.spacing,
          headerHeight: _headerHeight),
      Zoom.days => groupedGridSlivers(context,
          groups: groupByDay(repo.all, long: true),
          columns: _zoom.columns,
          spacing: _zoom.spacing,
          headerHeight: _headerHeight,
          thumbSize: 400),
    };

    return Stack(
      children: [
        Listener(
          onPointerDown: _onPointerDown,
          onPointerMove: _onPointerMove,
          onPointerUp: _onPointerUp,
          onPointerCancel: _onPointerUp,
          child: RefreshIndicator(
            edgeOffset: top,
            onRefresh: repo.load,
            child: Scrollbar(
              controller: _scroll,
              interactive: true,
              thickness: 6,
              radius: const Radius.circular(3),
              child: CustomScrollView(
                key: ValueKey(_zoom),
                controller: _scroll,
                physics: const AlwaysScrollableScrollPhysics(),
                slivers: [
                  SliverToBoxAdapter(child: SizedBox(height: top - 40)),
                  ...slivers,
                  SliverToBoxAdapter(
                    child: Padding(
                      padding: EdgeInsets.fromLTRB(16, 24, 16, bottom + 80),
                      child: Center(
                        child: Text(countLabel(repo.all),
                            style: TextStyle(
                                color: GalleryColors.subtle(
                                    Theme.of(context).brightness))),
                      ),
                    ),
                  ),
                ],
              ),
            ),
          ),
        ),
        Positioned(
          left: 24,
          right: 24,
          bottom: bottom + 12,
          child: Center(child: _ZoomSwitcher(value: _zoom, onChanged: _setZoom)),
        ),
      ],
    );
  }

  Widget _yearsSliver(List<AssetEntity> all) {
    final years = groupByYear(all).entries.toList()
      ..sort((a, b) => b.key.compareTo(a.key));
    final months = groupByMonth(all);

    return SliverPadding(
      padding: const EdgeInsets.fromLTRB(12, 40, 12, 0),
      sliver: SliverList.separated(
        itemCount: years.length,
        separatorBuilder: (_, __) => const SizedBox(height: 14),
        itemBuilder: (context, i) {
          final year = years[i];
          final preview = year.value.length <= 8
              ? year.value
              : List.generate(8, (k) => year.value[(k * year.value.length) ~/ 8]);
          return GestureDetector(
            onTap: () {
              final first = months.indexWhere(
                  (g) => g.assets.first.createDateTime.year == year.key);
              _setZoom(Zoom.months, jumpToGroup: math.max(0, first));
            },
            child: ClipRRect(
              borderRadius: BorderRadius.circular(22),
              child: SizedBox(
                height: 220,
                child: Stack(fit: StackFit.expand, children: [
                  Row(children: [
                    Expanded(flex: 2, child: AssetThumb(preview.first, size: 500, showBadges: false)),

class _SearchScreenState extends State<SearchScreen> {

  Timer? _debounce;

  @override
  void dispose() {
    _debounce?.cancel();
    _text.dispose();
    super.dispose();
  }

  void _onChanged(String q) {
    _debounce?.cancel();
    _debounce = Timer(const Duration(milliseconds: 250), () => _run(q));
  }

  void _run(String q) {
    final repo = GalleryScope.of(context, listen: false);
    setState(() {
      _query = q.trim();
      _results = _query.isEmpty ? const [] : repo.search(_query);
    });
  }

  void _useSuggestion(String s) {
    _text.text = s;
    _text.selection = TextSelection.collapsed(offset: s.length);
    _run(s);
  }

  @override

                    radius: 26,

                                },

                    ),
                  ),
                ),

/// Plays a video only while its page is the active one.
class _VideoPage extends StatefulWidget {
  const _VideoPage({
    super.key,
    required this.asset,
    required this.active,
    required this.onTap,
  });

  final AssetEntity asset;
  final bool active;
  final VoidCallback onTap;

  State<_VideoPage> createState() => _VideoPageState();

class _VideoPageState extends State<_VideoPage> {
  VideoPlayerController? _controller;
  bool _starting = false;

  @override
  void initState() {
    super.initState();
    if (widget.active) _start();
  }

  @override
  void didUpdateWidget(covariant _VideoPage old) {
    super.didUpdateWidget(old);
    if (widget.active && !old.active) _start();
    if (!widget.active && old.active) _stop();
  }

  Future<void> _start() async {
    if (_controller != null || _starting) return;
    _starting = true;
    final File? file = await widget.asset.file;
    if (file == null || !mounted) {
      _starting = false;
      return;
    }
    final c = VideoPlayerController.file(file);
    await c.initialize();
    _starting = false;
    if (!mounted || !widget.active) {
      await c.dispose();
      return;
    }
    await c.setLooping(true);
    await c.play();
    c.addListener(() {
      if (mounted) setState(() {});
    });
    setState(() => _controller = c);
  }

  void _stop() {
    final c = _controller;
    _controller = null;
    c?.dispose();
    if (mounted) setState(() {});
  }

  @override
  void dispose() {
    _controller?.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final c = _controller;
    final ready = c != null && c.value.isInitialized;
    return GestureDetector(
      onTap: () {
        if (ready) c.value.isPlaying ? c.pause() : c.play();
        widget.onTap();
      },
      child: Stack(alignment: Alignment.center, children: [
        if (ready)
          Center(
            child: AspectRatio(
              aspectRatio: c.value.aspectRatio,
              child: VideoPlayer(c),
            ),
          )
        else
          Center(
            child: AspectRatio(
              aspectRatio: widget.asset.width > 0 && widget.asset.height > 0
                  ? widget.asset.width / widget.asset.height
                  : 16 / 9,
              child: AssetThumb(widget.asset, size: 800, showBadges: false),
            ),
          ),
        if (ready && !c.value.isPlaying)
          IconButton.filledTonal(
            iconSize: 40,
            onPressed: () => c.value.isPlaying ? c.pause() : c.play(),
            icon: Icon(c.value.isPlaying ? Icons.pause : Icons.play_arrow),
            style: IconButton.styleFrom(
                backgroundColor: Colors.black26, foregroundColor: Colors.white),
          ),
        if (ready)
          Positioned(
            left: 16,
            right: 16,
            bottom: MediaQuery.paddingOf(context).bottom + 84,
            child: Row(children: [
              Text(formatDuration(c.value.position),
                  style: const TextStyle(color: Colors.white, fontSize: 12)),
              const SizedBox(width: 8),
              Expanded(
                child: VideoProgressIndicator(
                  c,
                  allowScrubbing: true,
                  colors: const VideoProgressColors(
                    playedColor: Colors.white,
                    bufferedColor: Colors.white38,
                    backgroundColor: Colors.white24,
                  ),
                ),
              ),
              const SizedBox(width: 8),
              Text(formatDuration(c.value.duration),
                  style: const TextStyle(color: Colors.white, fontSize: 12)),
            ]),
          ),
        if (!ready && widget.active)
          const CircularProgressIndicator(color: Colors.white),
      ]),
    );
  }
}
